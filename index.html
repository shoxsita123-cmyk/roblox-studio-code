Barcha Skriptlar Jamlanmasi (Paste & Play)

-- =======================================================
-- 1. DOUBLE JUMP + SHIFT SPRINT (StarterPlayerScripts -> LocalScript)
-- =======================================================
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")
local Players = game:GetService("Players")

local player = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local humanoid = character:WaitForChild("Humanoid")
local camera = workspace.CurrentCamera

-- Double Jump parametrlari
local canDoubleJump = false
local hasDoubleJumped = false
local DOUBLE_JUMP_POWER = 65

humanoid.StateChanged:Connect(function(oldState, newState)
    if newState == Enum.HumanoidStateType.Landed then
        canDoubleJump = false
        hasDoubleJumped = false
    elseif newState == Enum.HumanoidStateType.Freefall then
        task.wait(0.1)
        if not hasDoubleJumped then
            canDoubleJump = true
        end
    end
end)

UserInputService.JumpRequest:Connect(function()
    if canDoubleJump and not hasDoubleJumped then
        hasDoubleJumped = true
        canDoubleJump = false
        humanoid.JumpPower = DOUBLE_JUMP_POWER
        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

-- Shift Sprint parametrlari
local SPRINT_SPEED = 35
local SPRINT_FOV = 85
local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == Enum.KeyCode.LeftShift then
        humanoid.WalkSpeed = SPRINT_SPEED
        TweenService:Create(camera, tweenInfo, {FieldOfView = SPRINT_FOV}):Play()
    end
end)

UserInputService.InputEnded:Connect(function(input, gpe)
    if input.KeyCode == Enum.KeyCode.LeftShift then
        humanoid.WalkSpeed = 16
        TweenService:Create(camera, tweenInfo, {FieldOfView = 70}):Play()
    end
end)

    if humanoid and not isTouched then

        platform.CanCollide = false
        task.wait(3)
        platform.Transparency = 0
        platform.CanCollide = true
        isTouched = false
    end

--]]

1-Dars: Double Jump va Sprint skriptlarini qayerga qo'yish kerak?

Harakatga (Jump, Sprint, FOV) tegishli skriptlar har doim LocalScript bo'lishi va Roblox Studio'dagi StarterPlayer -> StarterPlayerScripts jildining ichiga joylashtirilishi shart!

-- StarterPlayer -> StarterPlayerScripts -> MovementScript (LocalScript)
local UserInputService = game:GetService("UserInputService")
local player = game.Players.LocalPlayer
local humanoid = player.Character:WaitForChild("Humanoid")

-- LeftShift bosilganda tez yugurish
UserInputService.InputBegan:Connect(function(input, gpe)
    if not gpe and input.KeyCode == Enum.KeyCode.LeftShift then
        humanoid.WalkSpeed = 35
    end
end)

UserInputService.InputEnded:Connect(function(input, gpe)
    if input.KeyCode == Enum.KeyCode.LeftShift then
        humanoid.WalkSpeed = 16
    end
end)
2-Dars: Parkour Obby bosqichlarini qanday yaratish kerak?

Qiziqarli Parkour yaratish uchun batut (Jump Pad), yo'qolib qoluvchi bloklar va Checkpoint (bosqich bayroqchalari) sistemasi bilan Lava bloki uyg'unlashtiriladi. Har bir blok ichiga mos ravishda tayyor generator skriptlarini joylashtiring.

-- Parkour Batut (Part -> Script)
local pad = script.Parent
pad.Touched:Connect(function(hit)
    local hrp = hit.Parent:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.Velocity = Vector3.new(hrp.Velocity.X, 130, hrp.Velocity.Z) -- Balandga otadi
    end
end)
3-Dars: Touch hodisasi nima (Part.Touched)?

Roblox Studio'da har qanday blok (Part) bilan o'yinchi to'qnashganda Part.Touched hodisasi ishga tushadi. Parkour tuzoqlari va teleportlar aynan shu orqali ishlaydi.

-- Blok Touch hodisasini tekshirish kodi (Part -> Script)
local part = script.Parent

part.Touched:Connect(function(hit)
    print("Blokga nimadir tegdi: " .. hit.Name)
    local character = hit.Parent
    local humanoid = character:FindFirstChildWhichIsA("Humanoid")
    if humanoid then
        print(character.Name .. " o'yinchisi blok ustiga qadadi!")
    end
end)
